from dataclasses import dataclass
from datetime import datetime
from typing import List

@dataclass
class Glyph:
    id: int          # numeric id, -1 = Pre‑phase
    name: str        # canonical glyph name
    phase: int       #  -1 .. 12, 13 = ∞
    frequency: float
    decay: float

GLYPHS: List[Glyph] = [
    Glyph(-1, "KIV-EEN",    -1, 0.8, 0.07),
    Glyph( 0, "EST-ONN",     0, 1.0, 0.05),
    Glyph( 1, "ØRU-KAI",     1, 1.2, 0.06),
    Glyph( 2, "VEH-TAL",     2, 1.3, 0.05),
    Glyph( 3, "ZUN-RAEK",    3, 1.4, 0.05),
    Glyph( 4, "KEL-TORUN",   4, 1.1, 0.06),
    Glyph( 5, "NAR-AETH",    5, 1.0, 0.05),
    Glyph( 6, "SHA-RUL",     6, 1.3, 0.04),
    Glyph( 7, "UTH-NAKH",    7, 1.2, 0.05),
    Glyph( 8, "DREZ-VUKH",   8, 1.1, 0.06),
    Glyph( 9, "VHEL-SURIK",  9, 1.4, 0.05),
    Glyph(10, "KAI-ELUN",   10, 1.2, 0.05),
    Glyph(11, "RHI-TUUM",   11, 1.0, 0.05),

    Glyph(13, "SEY-MOOR",   13, 0.9, 0.03),   # ∞‑phase “Witness”
]

def get_active_glyph() -> Glyph:
    """
    Return the glyph considered 'active' right now.
    Simple daily cycling based on UTC day of the month.
    """
    day_index = int(datetime.utcnow().strftime("%d")) % len(GLYPHS)
    return GLYPHS[day_index]
