import math
import matplotlib.pyplot as plt
import matplotlib.animation as animation

class Glyph:
    def __init__(self, name, phase, pos, ritual_meaning, energy=1.0):
        self.name = name
        self.phase = phase
        self.x, self.y = pos
        self.ritual_meaning = ritual_meaning
        self.energy = energy
        self.is_active = False
        self.subglyphs = []
        self.connections = []

    def add_subglyph(self, glyph):
        self.subglyphs.append(glyph)

    def add_connection(self, glyph):
        self.connections.append(glyph)

    def activate(self):
        self.is_active = True
        self.energy = min(self.energy * 1.5, 3.0)

    def deactivate(self):
        self.is_active = False
        self.energy = max(self.energy * 0.7, 0.2)


def generate_spiral_grid(num_points, spacing=1.5):
    points = []
    golden_angle = math.pi * (3 - math.sqrt(5))  # ≈2.399963
    for n in range(num_points):
        r = spacing * math.sqrt(n)
        theta = n * golden_angle
        x = r * math.cos(theta)
        y = r * math.sin(theta)
        points.append((x, y))
    return points


def create_lunareth_glyphs(names):
    ritual_meanings = [
        "Origin", "Ignition", "Orbit", "Fracture", "Reflection",
        "Echo", "Inversion", "Memory", "Mutation", "Radiation",
        "Absorption", "Transcendence", "Unity"
    ]
    points = generate_spiral_grid(len(names) * 2)
    glyphs = []

    for i, name in enumerate(names):
        g = Glyph(name, i, points[i], ritual_meanings[i])
        # Create 3 subglyphs for recursion demo
        for j in range(3):
            angle = j * (2 * math.pi / 3)
            sub_x = g.x + 0.3 * math.cos(angle)
            sub_y = g.y + 0.3 * math.sin(angle)
            sg = Glyph(f"{name}-sub{j+1}", i, (sub_x, sub_y), f"{ritual_meanings[i]} Aspect {j+1}", energy=0.5)
            g.add_subglyph(sg)
        glyphs.append(g)

    # Connect glyphs sequentially
    for i in range(1, len(glyphs)):
        glyphs[i-1].add_connection(glyphs[i])

    return glyphs


def update_energies(glyphs, time_step):
    active_index = time_step % len(glyphs)
    for i, g in enumerate(glyphs):
        if i == active_index:
            g.activate()
        else:
            g.deactivate()
        # Subglyphs reflect parent energy partially
        for sg in g.subglyphs:
            sg.energy = g.energy * 0.6 if g.is_active else max(sg.energy * 0.7, 0.1)
            sg.is_active = g.is_active


def plot_glyph(ax, glyph, depth=0):
    base_size = 200 / (depth + 1)
    size = base_size * glyph.energy
    color_hue = (glyph.phase * 30) % 360
    # Convert HSL hue to RGB approximation
    # Using matplotlib colormap instead of direct HSL for simplicity
    cmap = plt.get_cmap('hsv')
    color = cmap(color_hue / 360)
    alpha = 0.9 if glyph.is_active else 0.3

    circle = plt.Circle((glyph.x, glyph.y), radius=size * 0.02, color=color, alpha=alpha, ec='black' if glyph.is_active else 'gray', lw=1.2)
    ax.add_patch(circle)

    if depth == 0:
        ax.text(glyph.x, glyph.y - size * 0.03, glyph.name, fontsize=8, ha='center', color='white' if glyph.is_active else 'gray')

    for sg in glyph.subglyphs:
        plot_glyph(ax, sg, depth + 1)


def draw_connections(ax, glyphs):
    for g in glyphs:
        for conn in g.connections:
            ax.plot([g.x, conn.x], [g.y, conn.y], color='lightblue', lw=0.7, alpha=0.6)


def animate(i, glyphs, ax):
    ax.clear()
    ax.set_facecolor('black')
    ax.axis('off')
    ax.set_aspect('equal')
    ax.autoscale_view()

    update_energies(glyphs, i)
    draw_connections(ax, glyphs)

    for g in glyphs:
        plot_glyph(ax, g)


def main():
    glyph_names = [
        "EST-ONN", "ØRU-KAI", "VEH-TAL", "ZUN-RAEK", "KEL-TORUN",
        "NAR-AETH", "SHA-RUL", "UTH-NAKH", "DREZ-VUKH", "VHEL-SURIK",
        "KAI-ELUN", "RHI-TUUM", "XAH-MORU"
    ]

    glyphs = create_lunareth_glyphs(glyph_names)

    fig, ax = plt.subplots(figsize=(8, 8))
    ani = animation.FuncAnimation(fig, animate, fargs=(glyphs, ax), interval=500)
    plt.show()


if __name__ == "__main__":
    main()
