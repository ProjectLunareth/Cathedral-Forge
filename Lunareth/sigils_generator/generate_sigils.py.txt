import os
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Circle, Polygon, Arc, Ellipse, FancyArrow
from matplotlib.lines import Line2D

OUTPUT_FOLDER = "sigils_generator/output/"
os.makedirs(OUTPUT_FOLDER, exist_ok=True)

def save_fig(fig, filename):
    path = os.path.join(OUTPUT_FOLDER, filename)
    fig.savefig(path, dpi=300, transparent=True)
    plt.close(fig)

def setup_canvas():
    fig, ax = plt.subplots(figsize=(2, 2))
    ax.set_xlim(-1, 1)
    ax.set_ylim(-1, 1)
    ax.axis('off')
    return fig, ax

# ──────── SIGIL DEFINITIONS ─────────

def sigil_neg1_kiv_een():
    fig, ax = setup_canvas()
    np.random.seed(42)
    for _ in range(120):
        x, y = np.random.normal(0, 0.4, 2)
        size = np.random.uniform(0.01, 0.05)
        alpha = np.random.uniform(0.1, 0.5)
        ax.add_patch(Circle((x, y), size, color='white', alpha=alpha))
    ax.add_patch(Circle((0, 0), 0.2, edgecolor='white', facecolor='none', lw=1, alpha=0.3))
    save_fig(fig, "-1_KIV-EEN.png")

def sigil_0_est_onn():
    fig, ax = setup_canvas()
    arc1 = Arc((-0.4, 0), 0.5, 1.0, theta1=90, theta2=270, lw=2, color='white')
    arc2 = Arc(( 0.4, 0), 0.5, 1.0, theta1=270, theta2=90, lw=2, color='white')
    eye = Circle((0, 0), 0.05, color='white')
    ax.add_patch(arc1)
    ax.add_patch(arc2)
    ax.add_patch(eye)
    save_fig(fig, "0_EST-ONN.png")

def sigil_1_veh_tal():
    fig, ax = setup_canvas()
    triangle = Polygon([[-0.3, -0.3], [0.3, -0.3], [0, 0.4]], closed=True, edgecolor='white', facecolor='none', lw=2)
    dot = Circle((0, 0), 0.05, color='white')
    ax.add_patch(triangle)
    ax.add_patch(dot)
    save_fig(fig, "1_VEH-TAL.png")

def sigil_2_zun_raek():
    fig, ax = setup_canvas()
    arc1 = Arc((-0.25, 0), 0.5, 1.0, theta1=90, theta2=270, lw=2, color='white')
    arc2 = Arc((0.25, 0), 0.5, 1.0, theta1=270, theta2=90, lw=2, color='white')
    node = Circle((0, 0), 0.07, color='white')
    ax.add_patch(arc1)
    ax.add_patch(arc2)
    ax.add_patch(node)
    save_fig(fig, "2_ZUN-RAEK.png")

def sigil_3_kel_torun():
    fig, ax = setup_canvas()
    ax.add_line(Line2D([-0.8, 0.8], [-0.8, 0.8], color='white', lw=2))
    ax.add_line(Line2D([-0.8, 0.8], [0.8, -0.8], color='white', lw=2))
    save_fig(fig, "3_KEL-TORUN.png")

def sigil_4_nar_aeth():
    fig, ax = setup_canvas()
    ax.add_line(Line2D([0, 0], [-1, 1], color='white', lw=2))
    ax.add_patch(Ellipse((0, 0.4), 0.8, 0.3, edgecolor='white', facecolor='none', lw=1))
    ax.add_patch(Ellipse((0, -0.4), 0.8, 0.3, edgecolor='white', facecolor='none', lw=1))
    save_fig(fig, "4_NAR-AETH.png")

def sigil_5_sha_rul():
    fig, ax = setup_canvas()
    ax.add_patch(Arc((0, 0), 1.0, 1.0, theta1=0, theta2=180, lw=2, color='white'))
    ax.add_patch(Arc((0, 0), 0.7, 0.7, theta1=0, theta2=180, lw=1.5, color='white', alpha=0.7))
    ax.add_patch(Arc((0, 0), 0.4, 0.4, theta1=0, theta2=180, lw=1, color='white', alpha=0.4))
    save_fig(fig, "5_SHA-RUL.png")

def sigil_6_uth_nakh():
    fig, ax = setup_canvas()
    outer = Polygon([[0, -0.5], [-0.5, 0.5], [0.5, 0.5]], closed=True, edgecolor='white', facecolor='none', lw=2)
    inner = Polygon([[0, -0.3], [-0.3, 0.3], [0.3, 0.3]], closed=True, edgecolor='white', facecolor='none', lw=1)
    ax.add_patch(outer)
    ax.add_patch(inner)
    save_fig(fig, "6_UTH-NAKH.png")

def sigil_7_drez_vukh():
    fig, ax = setup_canvas()
    for r in [0.7, 0.5, 0.3, 0.1]:
        ax.add_patch(Arc((0, 0), r, r, theta1=0, theta2=360, lw=1.5-(r*1.5), color='white'))
    save_fig(fig, "7_DREZ-VUKH.png")

def sigil_8_vhel_surik():
    fig, ax = setup_canvas()
    ax.add_line(Line2D([-0.7, 0.7], [-0.7, 0.7], color='white', lw=2))
    ax.add_line(Line2D([-0.7, 0.7], [0.7, -0.7], color='white', lw=2))
    for i in range(1, 4):
        r = i * 0.2
        ax.add_patch(Circle((0, 0), r, edgecolor='white', facecolor='none', alpha=0.3))
    save_fig(fig, "8_VHEL-SURIK.png")

def sigil_9_kai_elun():
    fig, ax = setup_canvas()
    core = Circle((0, 0), 0.1, color='white')
    ax.add_patch(core)
    for angle in range(0, 360, 30):
        dx = 0.5 * np.cos(np.radians(angle))
        dy = 0.5 * np.sin(np.radians(angle))
        ax.add_line(Line2D([0, dx], [0, dy], color='white', lw=1.5))
    save_fig(fig, "9_KAI-ELUN.png")

def sigil_10_rhi_tuum():
    fig, ax = setup_canvas()
    for i in range(-1, 2, 2):
        ax.add_patch(FancyArrow(i*0.5, 0, -i*0.4, 0, width=0.03, color='white'))
    ax.add_patch(Circle((0, 0), 0.07, color='white'))
    save_fig(fig, "10_RHI-TUUM.png")

def sigil_11_xah_moru():
    fig, ax = setup_canvas()
    t = np.linspace(-1, 1, 200)
    x1 = 0.2 * np.sin(4 * np.pi * t)
    x2 = -0.2 * np.sin(4 * np.pi * t)
    ax.plot(x1, t, color='white', lw=2)
    ax.plot(x2, t, color='white', lw=2)
    ax.add_patch(Circle((0, 1.05), 0.05, color='white', alpha=0.3))
    ax.add_patch(Circle((0, -1.05), 0.05, color='white', alpha=0.3))
    save_fig(fig, "11_XAH-MORU.png")

def sigil_12_final():
    fig, ax = setup_canvas()
    ax.add_patch(Circle((0, 0), 0.9, edgecolor='white', facecolor='none', lw=2))
    ax.add_patch(Polygon([[0, 0.9], [-0.6, -0.5], [0.6, -0.5]], closed=True, edgecolor='white', facecolor='none', lw=2))
    ax.add_line(Line2D([-0.6, 0.6], [0, 0], color='white', lw=1))
    save_fig(fig, "12_FINAL.png")
